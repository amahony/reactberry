import {Playground} from 'docz';
import {
  Box,
  GlobalStyle,
  Heading,
  Text,
  ThemeProvider,
  defaultTheme
} from '../..';

export function Colormap() {
  return (
    <Box display="grid" gridGap="small" gridTemplateColumns="repeat(8, 1fr)">
      {Object.keys(defaultTheme.colors).map(
        (color, i) =>
          color !== 'palette' &&
          color !== 'intents' && (
            <Box key={i} border="1px solid" borderColor="light" shadow="small">
              <Box bg={color} p="2rem" />
              <Heading fontSize="xsmall" px="xsmall">
                {color}
              </Heading>
            </Box>
          )
      )}
    </Box>
  );
}

export function Palettemap() {
  return (
    <Box>
      {Object.keys(defaultTheme.colors.palette).map((item, i) => (
        <Box key={i}>
          <Heading fontSize="medium" mb={0}>
            {item}
          </Heading>
          <Box
            display="grid"
            gridGap="small"
            gridTemplateColumns="repeat(10, 1fr)"
          >
            {Object.values(defaultTheme.colors.palette[item]).map((data, i) => (
              <Box flex="1 0 auto" key={i}>
                <Heading fontSize="xsmall">
                  {item}.{i}
                </Heading>
                <Box bg={data} p="2rem" />
                <Text fontSize="xsmall">{data}</Text>
              </Box>
            ))}
          </Box>
        </Box>
      ))}
    </Box>
  );
}

export function Intentsmap() {
  return (
    <Box>
      {Object.keys(defaultTheme.colors.intents).map((item, i) => (
        <Box key={i}>
          <Heading fontSize="medium" mb={0}>
            {item}
          </Heading>
          <Box
            display="grid"
            gridGap="small"
            gridTemplateColumns="repeat(10, 1fr)"
          >
            {Object.values(defaultTheme.colors.intents[item]).map((data, i) => (
              <Box flex="1 0 auto" key={i}>
                <Heading fontSize="xsmall">
                  {item}.{i}
                </Heading>
                <Box bg={data} p="2rem" />
                <Text fontSize="xsmall">{data}</Text>
              </Box>
            ))}
          </Box>
        </Box>
      ))}
    </Box>
  );
}

# Theme

This page documents the current Reactberry theme contract within the `@reactberry/core` docs.

This repo-local page mirrors the supported public theme contract for the first release: use `@reactberry/core/theme` for `ThemeProvider`, `defaultTheme`, and `GlobalStyle`.

This contract remains part of the hard-break migration posture, so consumer guidance stays the same: use supported package entries only and do not rely on `/src` or `/dist` deep imports.
